  <script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script>
  <script>
    document.addEventListener('DOMContentLoaded', async () => {

      /* ---------- 1. КАРТА ---------- */
      const map = L.map('map').setView([55.7558, 37.6176], 10);
      L.tileLayer('https://core-renderer-tiles.maps.yandex.net/tiles?l=map&v=21.06.28&x={x}&y={y}&z={z}&scale=1&lang=ru_RU&projection=web_mercator', {
        attribution: '© Яндекс'
      }).addTo(map);
      window.addEventListener('resize', () => map.invalidateSize());
      setTimeout(() => map.invalidateSize(), 300);

      /* ---------- 2. ЭЛЕМЕНТЫ UI ---------- */
      const sidebar      = document.querySelector('.sidebar');
      const sidebarTitle = sidebar.querySelector('h3');
      const sidebarText  = sidebar.querySelector('p');
      const picturebar   = document.querySelector('.picturebar');
      const siteTitleEl   = document.getElementById('siteTitle');
      const loaderEl      = document.getElementById('loader');
      const loaderToggle  = document.getElementById('loaderToggle');
      const sheetUrlInput = document.getElementById('sheetUrl');
      const loadBtn       = document.getElementById('loadBtn');
      const loaderMsg     = document.getElementById('loaderMsg');

      let locations = [];
      let markers   = [];
      let current   = 0;
      let loading   = false;

      const DEFAULT_SHEET_ID   = '1TsdpNmFmBS-3IdxjCSl3jGksqVFkFGGZvikmgpaoy-Y';
      const DEFAULT_SITE_TITLE = 'История восстания декабристов';

      /* ---------- 2b. ЗАГОЛОВОК САЙТА = ЗАГОЛОВОК ТАБЛИЦЫ ---------- */
      function setSiteTitle(title) {
        const text = (title && String(title).trim()) || DEFAULT_SITE_TITLE;
        if (siteTitleEl) siteTitleEl.textContent = text;
        document.title = text;
      }

      /* !!! ВСТАВЬТЕ СВОЙ API-КЛЮЧ ЗДЕСЬ !!! */
      const GOOGLE_API_KEY = 'ВАШ_API_КЛЮЧ';

      /**
       * Получает название таблицы через Google Sheets API v4.
       * Использует fields=properties.title, чтобы запросить только заголовок.
       * @param {string} sheetId - ID таблицы.
       * @returns {Promise<string|null>} Название таблицы или null.
       */
      async function fetchSheetTitle(sheetId) {
        if (!GOOGLE_API_KEY || GOOGLE_API_KEY === 'ВАШ_API_КЛЮЧ') {
          console.warn('API-ключ не указан. Название таблицы не будет загружено через API.');
          return null;
        }
        try {
          const url = `https://sheets.googleapis.com/v4/spreadsheets/${sheetId}?fields=properties.title&key=${GOOGLE_API_KEY}`;
          const resp = await fetch(url);
          if (!resp.ok) {
            console.warn(`Sheets API вернул ${resp.status} для ${sheetId}.`);
            return null;
          }
          const data = await resp.json();
          return data?.properties?.title || null;
        } catch (err) {
          console.warn('Ошибка при запросе к Sheets API:', err);
          return null;
        }
      }

      /* ---------- 2c. СВОРАЧИВАНИЕ ПАНЕЛИ ---------- */
      function setLoaderCollapsed(collapsed) {
        loaderEl.classList.toggle('collapsed', collapsed);
        loaderToggle.textContent = collapsed ? '+' : '−';
        loaderToggle.title       = collapsed ? 'Развернуть' : 'Свернуть';
      }
      loaderToggle.addEventListener('click', () => {
        setLoaderCollapsed(!loaderEl.classList.contains('collapsed'));
      });

      /* ---------- 3. CSV ---------- */
      function parseCSV(csv) {
        const rows = [];
        let row = [], field = '', inQuotes = false;
        for (let i = 0; i < csv.length; i++) {
          const ch = csv[i], nx = csv[i + 1];
          if (ch === '"') {
            if (inQuotes && nx === '"') { field += '"'; i++; }
            else inQuotes = !inQuotes;
          } else if (ch === ',' && !inQuotes) {
            row.push(field.trim()); field = '';
          } else if ((ch === '\n' || ch === '\r') && !inQuotes) {
            if (ch === '\r' && nx === '\n') i++;
            row.push(field.trim()); rows.push(row);
            row = []; field = '';
          } else {
            field += ch;
          }
        }
        if (field || row.length) { row.push(field.trim()); rows.push(row); }
        return rows;
      }

      /* ---------- 4. КООРДИНАТЫ ---------- */
      function parseCoordinates(cell) {
        if (!cell) return [];
        const coords = [];
        const cleaned = cell.trim().replace(/^[\[\(]|[\]\)]$/g, '').trim();
        const re = /(-?\d+(?:\.\d+)?)[,\s]+(-?\d+(?:\.\d+)?)/g;
        let m;
        while ((m = re.exec(cleaned)) !== null) {
          const lat = parseFloat(m[1]), lon = parseFloat(m[2]);
          if (!isNaN(lat) && !isNaN(lon) && lat >= -90 && lat <= 90 && lon >= -180 && lon <= 180) {
            coords.push([lat, lon]);
          }
        }
        return coords;
      }

      /* ---------- 4b. ИЗВЛЕЧЕНИЕ ID ИЗ ССЫЛКИ ---------- */
      function extractSheetId(input) {
        if (!input) return null;
        const trimmed = input.trim();
        if (/^[a-zA-Z0-9-_]{20,}$/.test(trimmed)) return trimmed;
        const m = trimmed.match(/\/spreadsheets\/d\/(?!e\/)([a-zA-Z0-9-_]+)/);
        return m ? m[1] : null;
      }

      /* ---------- 5. ЗАГРУЗКА ДАННЫХ ---------- */
      async function loadAllData(sheetId) {
        if (!sheetId) throw new Error('Не указан ID таблицы');

        markers.forEach(m => m.remove());
        markers = [];
        locations = [];
        current = 0;
        loading = false;

        sidebarTitle.innerText = 'Загрузка…';
        sidebarText.innerText  = 'Подключение к таблице';
        picturebar.innerHTML   = '<h3>Загрузка…</h3><p>Подключение к таблице</p>';
        setSiteTitle('Загрузка…');

        // Запрашиваем CSV и заголовок через API параллельно
        const titlePromise = fetchSheetTitle(sheetId);

        const url = `https://docs.google.com/spreadsheets/d/${sheetId}/gviz/tq?tqx=out:csv`;
        const resp = await fetch(url);
        if (!resp.ok) throw new Error('HTTP ' + resp.status);

        const csv  = await resp.text();
        const rows = parseCSV(csv);
        if (rows.length < 2) throw new Error('Нет данных в таблице');

        const headers = rows[0];
        let coordCol = headers.findIndex(h => h.toLowerCase().includes('coord') || h.toLowerCase().includes('координат'));
        if (coordCol === -1) coordCol = 7;
        let picCol = headers.findIndex(h => {
          const l = h.toLowerCase();
          return l.includes('picture') || l.includes('картинка') || l.includes('изображение');
        });

        for (let i = 1; i < rows.length; i++) {
          const row = rows[i];
          if (!row || row.length === 0) continue;
          if (!row.some(c => c && c.trim().length > 0)) continue;
          const coords = parseCoordinates(row[coordCol] || '');
          if (coords.length === 0) continue;
          const title    = (row[1] || row[2] || `Локация ${i}`).trim();
          const desc     = (row[3] || '').trim();
          const place    = (row[4] || '').trim();
          const zoom     = Math.min(18, Math.max(5, parseInt(row[5], 10) || 16));
          const picture  = (picCol !== -1 && row[picCol]) ? row[picCol].trim() : '';
          locations.push({ coordsList: coords, zoom, title, desc: `${desc} ${place ? `(${place})` : ''}`.trim(), picture });
        }

        if (locations.length === 0) {
          throw new Error('В таблице не найдено ни одной локации с координатами');
        }

        // Устанавливаем заголовок сайта
        const apiTitle = await titlePromise;
        const isDefault = sheetId === DEFAULT_SHEET_ID;
        let tableTitle;

        if (apiTitle) {
          // Успешно получено настоящее имя таблицы через Sheets API v4
          tableTitle = apiTitle;
        } else if (isDefault) {
          tableTitle = DEFAULT_SITE_TITLE;
        } else {
          // Резервный вариант: если API-ключ не указан или запрос не удался,
          // показываем название первой локации.
          tableTitle = (locations[0] && locations[0].title) || 'Своя таблица';
        }
        setSiteTitle(tableTitle);

        sidebarTitle.innerText = `${locations.length} локаций`;
        sidebarText.innerText  = 'Колёсико мыши / стрелки ↑↓';
        goTo(0);
      }

      /* ---------- 6. ПОКАЗ ЛОКАЦИИ ---------- */
      function goTo(index) {
        if (!locations[index] || loading) return;
        loading = true;
        const loc = locations[index];
        markers.forEach(m => m.remove());
        markers = [];
        const colors = ['red','orange','green','blue','violet','grey','black','gold'];
        loc.coordsList.forEach((c, cIdx) => {
          const color = colors[cIdx % colors.length];
          const icon = L.icon({
            iconUrl:    `https://raw.githubusercontent.com/pointhi/leaflet-color-markers/master/img/marker-icon-2x-${color}.png`,
            shadowUrl:  'https://cdnjs.cloudflare.com/ajax/libs/leaflet/0.7.7/images/marker-shadow.png',
            iconSize:   [25, 41], iconAnchor: [12, 41], popupAnchor:[1, -34], shadowSize: [41, 41]
          });
          const marker = L.marker(c, { icon }).bindPopup(`<b>${loc.title}</b><br>${loc.desc.substring(0, 200)}`, { autoPan: false });
          marker.addTo(map);
          markers.push(marker);
        });
        const target = loc.coordsList[0];
        map.flyTo(target, loc.zoom, { animate: true, duration: 1 });
        sidebarTitle.innerText = loc.title;
        sidebarText.innerText  = loc.desc || 'Описание отсутствует';
        setTimeout(() => { if (markers[0]) markers[0].openPopup(); }, 300);
        picturebar.innerHTML = loc.picture ? `<img src="${loc.picture}" alt="">` : `<h3>Нет изображения</h3><p>Колонка picture пуста</p>`;
        current = index;
        loading = false;
      }

      /* ---------- 7. НАВИГАЦИЯ ---------- */
      sidebar.addEventListener('wheel', (e) => {
        if (locations.length === 0) return;
        e.preventDefault();
        const delta = e.deltaY > 0 ? 1 : -1;
        let next = current + delta;
        if (next < 0) next = 0;
        if (next >= locations.length) next = locations.length - 1;
        if (next !== current) goTo(next);
      }, { passive: false });
      sidebar.addEventListener('mouseenter', () => sidebar.focus());
      document.addEventListener('keydown', (e) => {
        if (e.key === 'ArrowUp' || e.key === 'ArrowDown') {
          e.preventDefault();
          const delta = e.key === 'ArrowDown' ? 1 : -1;
          let next = current + delta;
          if (next < 0) next = 0;
          if (next >= locations.length) next = locations.length - 1;
          if (next !== current) goTo(next);
        }
      });

      /* ---------- 8. ЗАГРУЗКА ИЗ ПОЛЯ ВВОДА ---------- */
      async function loadFromInput() {
        const id = extractSheetId(sheetUrlInput.value);
        if (!id) {
          loaderMsg.textContent = 'Не удалось извлечь ID таблицы из ссылки';
          loaderMsg.className = 'loader-msg error';
          return;
        }
        loadBtn.disabled = true;
        loaderMsg.textContent = 'Загрузка…';
        loaderMsg.className = 'loader-msg';
        try {
          await loadAllData(id);
          loaderMsg.textContent = 'Данные загружены';
          loaderMsg.className = 'loader-msg ok';
          setLoaderCollapsed(true);
        } catch (err) {
          console.error(err);
          loaderMsg.textContent = err.message + '. Проверь, что доступ к таблице — «Все, у кого есть ссылка».';
          loaderMsg.className = 'loader-msg error';
          sidebarTitle.innerText = 'Ошибка загрузки';
          sidebarText.innerText  = err.message;
          picturebar.innerHTML   = '<h3>Нет данных</h3><p>Не удалось загрузить таблицу</p>';
        } finally {
          loadBtn.disabled = false;
        }
      }

      loadBtn.addEventListener('click', loadFromInput);
      sheetUrlInput.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') loadFromInput();
      });

      /* ---------- 9. СТАРТОВАЯ ЗАГРУЗКА ---------- */
      try {
        await loadAllData(DEFAULT_SHEET_ID);
      } catch (err) {
        console.error(err);
        sidebarTitle.innerText = 'Ошибка загрузки';
        sidebarText.innerText  = err.message + '. Проверь ID таблицы и доступ (File → Share → Anyone with the link).';
        picturebar.innerHTML   = '<h3>Нет данных</h3><p>Проверь доступ к таблице</p>';
        setSiteTitle(DEFAULT_SITE_TITLE);
      }

      setTimeout(() => sidebar.focus(), 500);
    });
  </script>
